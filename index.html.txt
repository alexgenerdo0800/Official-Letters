    async function deriveKey(salt) {
        const enc = new TextEncoder();
        const baseKey = await window.crypto.subtle.importKey(
            "raw", 
            enc.encode(HIDDEN_PIN), 
            { name: "PBKDF2" }, 
            false, 
            ["deriveKey"]
        );
        return window.crypto.subtle.deriveKey(
            {
                name: "PBKDF2",
                salt: salt,
                iterations: 100000,
                hash: "SHA-256"
            },
            baseKey,
            { name: "AES-GCM", length: 256 },
            false,
            ["encrypt", "decrypt"]
        );
    }
